import turtle
import math

def setup_turtle():
    """Initialize turtle with standard settings"""
    t = turtle.Turtle()
    t.speed(0)
    screen = turtle.Screen()
    screen.title("Turtle Graphics Assignment")
    return t, screen

def draw_rectangle(t, width, height, fill_color):
    if fill_color:
        t.fillcolor(fill_color)
        t.begin_fill()
    for _ in range(2):
        t.forward(width)
        t.right(90)
        t.forward(height)
        t.right(90)
    if fill_color:
        t.end_fill()

def draw_square(t, size, fill_color=None):
    if fill_color:
        t.fillcolor(fill_color)
        t.begin_fill()
    for _ in range(4):
        t.forward(size)
        t.right(90)
    if fill_color:
        t.end_fill()

def draw_triangle(t, size, fill_color=None):
    if fill_color:
        t.fillcolor(fill_color)
        t.begin_fill()
    for _ in range(3):
        t.forward(size)
        t.left(120)
    if fill_color:
        t.end_fill()

def draw_circle(t, radius, fill_color=None):
    if fill_color:
        t.fillcolor(fill_color)
        t.begin_fill()
    t.circle(radius)
    if fill_color:
        t.end_fill()

def draw_polygon(t, sides, size, fill_color=None):
    if fill_color:
        t.fillcolor(fill_color)
        t.begin_fill()
    angle = 360 / sides
    for _ in range(sides):
        t.forward(size)
        t.right(angle)
    if fill_color:
        t.end_fill()

def draw_curve(t, length, curve_factor, segments=10, fill_color=None):
    if fill_color:
        t.fillcolor(fill_color)
        t.begin_fill()
    segment_length = length / segments
    original_heading = t.heading()
    for i in range(segments):
        angle = curve_factor * math.sin(math.pi * i / segments)
        t.right(angle)
        t.forward(segment_length)
        t.left(angle)
    t.setheading(original_heading)
    if fill_color:
        t.end_fill()

def jump_to(t, x, y):
    t.penup()
    t.goto(x, y)
    t.pendown()

def draw_grass(t):
    draw_rectangle(t,800,300,"green")

def draw_house(t, x, y, size=150):
    jump_to(t, x, y)
    draw_square(t, size, fill_color="white")
    draw_triangle(t, size, fill_color="maroon")

    jump_to(t, x + 50, y - 70)
    draw_rectangle(t, 50, 80, fill_color="saddle brown")

    jump_to(t, x + 60, y - 120)
    draw_circle(t, 6, fill_color="wheat")

    jump_to(t, x + 105, y - 30)
    draw_square(t, 30, fill_color="white")

def draw_bush(t, x, y, radius=30, fill_color="forest green"):
    jump_to(t, x, y)
    draw_circle(t, radius, fill_color=fill_color)

def draw_tree(t, x, y, trunk_w=30, trunk_h=100, canopy_r=50):
    jump_to(t, x, y)
    draw_rectangle(t, trunk_w, trunk_h, fill_color="saddle brown")
    jump_to(t, x + trunk_w//2, y)
    draw_circle(t, canopy_r, fill_color="forest green")

def draw_cloud(t,x,y,width=120,height=60,fill_color="white"):
    jump_to(t,x,y)
    draw_rectangle(t,width,height,fill_color=fill_color)

def draw_sun(t,x,y,radius=90,fill_color="yellow"):
    jump_to(t,x,y)
    draw_circle(t,90,fill_color=fill_color)

def draw_scene(t):
    screen = t.getscreen()
    screen.bgcolor("skyblue")

    jump_to(t,-400,-200)
    draw_grass(t)

    draw_house(t,-75,-50)
    draw_house(t,-250,-50)
    draw_house(t,100,-50)

    draw_tree(t,320,-100)
    draw_tree(t,-350,-100)

    draw_bush(t,260,-200)
    draw_bush(t,90,-200)
    draw_bush(t,-90,-200)
    draw_bush(t,-260,-200)

    draw_cloud(t,-150,275)
    draw_cloud(t,30,350)
    draw_cloud(t,80,225)
    draw_cloud(t,-300,300)

    draw_sun(t,345,250)

def main():
    t, screen = setup_turtle()
    draw_scene(t)
    screen.mainloop()

if __name__ == "__main__":
    main()
